  <script>
    let timers = {
      team1: 300,
      team2: 240
    };
    let currentTeam = null;
    let interval = null;
    let thirtySecondAlertPlayed = {
      team1: false,
      team2: false
    };

    function startTimerSetup() {
      timers.team1 = parseInt(document.getElementById('team1Min').value) * 60 + parseInt(document.getElementById('team1Sec').value);
      timers.team2 = parseInt(document.getElementById('team2Min').value) * 60 + parseInt(document.getElementById('team2Sec').value);
      thirtySecondAlertPlayed.team1 = false;
      thirtySecondAlertPlayed.team2 = false;
      updateDisplay('team1');
      updateDisplay('team2');
      stopInterval();
      currentTeam = null;
    }

    function updateDisplay(team) {
      const seconds = timers[team];
      const min = Math.floor(seconds / 60).toString().padStart(2, '0');
      const sec = (seconds % 60).toString().padStart(2, '0');
      document.getElementById(`${team}Timer`).textContent = `${min}:${sec}`;
    }

    function stopInterval() {
      if (interval) {
        clearInterval(interval);
        interval = null;
      }
    }

    function pauseTimer() {
      stopInterval();
    }

    function resetTimers() {
      stopInterval();
      startTimerSetup();
    }

    function switchTeam(team) {
      stopInterval();
      currentTeam = team;
      interval = setInterval(() => {
        if (timers[team] > 0) {
          timers[team]--;
          updateDisplay(team);

          if (timers[team] === 30 && !thirtySecondAlertPlayed[team]) {
            document.getElementById('ding30').play();
            thirtySecondAlertPlayed[team] = true;
          }

          if (timers[team] === 0) {
            stopInterval();
            document.getElementById('ding0').play();
            alert(team + " 時間結束！");
          }
        }
      }, 1000);
    }
  </script>
